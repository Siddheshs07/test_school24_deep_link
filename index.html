3. Smart Deep Link (Recommended)

Tries to open app, falls back to store if not installed:



function smartDeepLink(path) {
  const universalLink = `https://app.school24.net.au${path}`;
  window.location.href = universalLink;
  
  setTimeout(() => {
    // Fallback to store if app didn't open
    redirectToStore();
  }, 2500);
}
